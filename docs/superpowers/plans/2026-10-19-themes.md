**Goal:** kb's colours come from base24 theme files in `~/.config/kb/themes/`, chosen live from More… → Theme… and
saved by name in `config.toml`; `kb-native` (kb's own colours) is the default.

**Architecture:** a new `theme` module reads scheme files (`scheme.rs`), maps base24 onto kb's 19 roles as a
`Palette`, checks legibility (`legibility.rs`), builds GPUI Kit's theme JSON in Rust (`gpui.rs`, the port of
`design:theme`), and owns the themes folder (`folder.rs`) and the active palette. `ui::design::design()` becomes a
swappable `Arc<Design>` whose `colors` the active palette sets. The action panel gains a Theme… submenu that
previews on selection and saves on ↵ through `toml_edit`.

**Tech Stack:** Rust, GPUI Kit 0.7.1 (`ThemeRegistry::load_themes_from_str`, `Theme::change`), saphyr through
`kb_domain::yaml`, `toml_edit`, cargo-nextest, Bun (one-off generator).

use gpui_kit::Hsla;

/// A colour as 8-bit channels.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Rgba8 {
    /// Red.
    pub r: u8,
    /// Green.
    pub g: u8,
    /// Blue.
    pub b: u8,
    /// Opacity, 255 opaque.
    pub a: u8,
}

/// A channel computed in `f64`, rounded half up and clamped as `color.ts`'s `toHex` and `tint` do.
fn channel(value: f64) -> u8 {
    #[expect(clippy::cast_possible_truncation, clippy::cast_sign_loss, reason = "clamped to 0..=255 first")]
    {
        value.round().clamp(0., 255.) as u8
    }
}

impl Rgba8 {
    /// `#RRGGBB` or `#RRGGBBAA`; the `#` is optional, as older base16 files omit it.
    #[must_use]
    pub fn parse(text: &str) -> Option<Self> {
        let digits = text.trim().trim_start_matches('#');
        let value = u32::from_str_radix(digits, 16).ok()?;
        match digits.len() {
            6 => {
                let [_, r, g, b] = value.to_be_bytes();
                Some(Self { r, g, b, a: 255 })
            }
            8 => {
                let [r, g, b, a] = value.to_be_bytes();
                Some(Self { r, g, b, a })
            }
            _ => None,
        }
    }

    /// `#RRGGBB` when opaque, else `#RRGGBBAA`, upper-case.
    #[must_use]
    pub fn hex(self) -> String {
        if self.a == 255 {
            format!("#{:02X}{:02X}{:02X}", self.r, self.g, self.b)
        } else {
            format!("#{:02X}{:02X}{:02X}{:02X}", self.r, self.g, self.b, self.a)
        }
    }

    /// The same colour at `alpha` (0–1).
    #[must_use]
    pub fn with_alpha(self, alpha: f64) -> Self {
        Self { a: channel(alpha * 255.), ..self }
    }

    fn channels(self) -> [f64; 3] {
        [f64::from(self.r), f64::from(self.g), f64::from(self.b)]
    }

    /// `self` drawn at `alpha` over `under`, opaque (`color.ts`'s `mix`).
    #[must_use]
    pub fn mix(self, under: Self, alpha: f64) -> Self {
        let (f, u) = (self.channels(), under.channels());
        let at = |i: usize| channel(f[i] * alpha + u[i] * (1. - alpha));
        Self { r: at(0), g: at(1), b: at(2), a: 255 }
    }

    /// The colour that, drawn at `alpha` over `under`, shows as `self` (`color.ts`'s `tint`), clamped to what can be
    /// drawn instead of failing: GPUI Kit caps some highlights' opacity.
    #[must_use]
    pub fn tint(self, under: Self, alpha: f64) -> Self {
        let (t, u) = (self.channels(), under.channels());
        let at = |i: usize| channel(u[i] + (t[i] - u[i]) / alpha);
        Self { r: at(0), g: at(1), b: at(2), a: channel(alpha * 255.) }
    }

    fn luminance(self) -> f64 {
        let lin = |v: f64| {
            let c = v / 255.;
            if c <= 0.04045 { c / 12.92 } else { ((c + 0.055) / 1.055).powf(2.4) }
        };
        let [r, g, b] = self.channels().map(lin);
        0.2126 * r + 0.7152 * g + 0.0722 * b
    }

    /// The WCAG 2.2 contrast ratio of two opaque colours.
    #[must_use]
    pub fn contrast(self, other: Self) -> f64 {
        let (x, y) = (self.luminance(), other.luminance());
        (x.max(y) + 0.05) / (x.min(y) + 0.05)
    }

    /// The colour GPUI draws.
    #[must_use]
    pub fn hsla(self) -> Hsla {
        gpui_kit::Rgba {
            r: f32::from(self.r) / 255.,
            g: f32::from(self.g) / 255.,
            b: f32::from(self.b) / 255.,
            a: f32::from(self.a) / 255.,
        }
        .into()
    }
}
```

use crate::theme::color::Rgba8;

/// kb's colour roles, in DESIGN.md's order.
pub const ROLES: [&str; 19] = [
    "window", "mantle", "crust", "selected", "hover", "badge", "code", "edge", "ring", "text", "muted", "faint",
    "mauve", "blue", "green", "peach", "pink", "yellow", "red",
];

/// A value for every role.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Palette([Rgba8; 19]);

impl Palette {
    fn index(role: &str) -> Option<usize> {
        ROLES.iter().position(|r| *r == role)
    }

    /// A role's colour; `None` for a name that is not a role.
    #[must_use]
    pub fn get(&self, role: &str) -> Option<Rgba8> {
        Self::index(role).map(|i| self.0[i])
    }

    /// A role's colour, for the roles this module knows exist.
    #[must_use]
    pub fn role(&self, role: &str) -> Rgba8 {
        self.get(role).unwrap_or(Rgba8 { r: 0, g: 0, b: 0, a: 255 })
    }

    fn set(&mut self, role: &str, value: Rgba8) -> bool {
        Self::index(role).map(|i| self.0[i] = value).is_some()
    }

    /// Overrides from a table of `role = "#hex"` (config.toml's `[colors]`); returns a notice per key it could
    /// not use.
    pub fn apply(&mut self, table: &toml::Table) -> Vec<String> {
        let mut notices = Vec::new();
        for (key, value) in table {
            let parsed = value.as_str().and_then(Rgba8::parse);
            match parsed {
                Some(colour) if self.set(key, colour) => {}
                Some(_) => notices.push(format!("Ignored colors.{key} in config.toml: not a colour role")),
                None => notices.push(format!("Ignored colors.{key} in config.toml: not #RRGGBB or #RRGGBBAA")),
            }
        }
        notices
    }

    /// The colours kb's views draw with.
    #[must_use]
    pub fn colors(&self) -> crate::ui::design::Colors {
        let c = |role: &str| self.role(role).hsla();
        crate::ui::design::Colors {
            window: c("window"),
            mantle: c("mantle"),
            crust: c("crust"),
            selected: c("selected"),
            hover: c("hover"),
            badge: c("badge"),
            code: c("code"),
            edge: c("edge"),
            ring: c("ring"),
            text: c("text"),
            muted: c("muted"),
            faint: c("faint"),
            mauve: c("mauve"),
            blue: c("blue"),
            green: c("green"),
            peach: c("peach"),
            pink: c("pink"),
            yellow: c("yellow"),
            red: c("red"),
        }
    }
}

/// A theme file, read.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Scheme {
    /// The file stem: what config.toml names.
    pub id: String,
    /// The file's `name`, else the id.
    pub label: String,
    /// Its colours.
    pub palette: Palette,
}

/// Why a file is not a theme.
#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
pub enum SchemeError {
    /// Not YAML.
    #[error("not YAML: {0}")]
    Yaml(String),
    /// A required colour is absent.
    #[error("no {0}")]
    Missing(String),
    /// A value is not a colour.
    #[error("{key} is not a colour: {value}")]
    BadColour {
        /// Where.
        key: String,
        /// What it held.
        value: String,
    },
}

/// base24's fallbacks for base16 files (base24 styling.md): base10/11 → base00, base12–17 → their normal colours.
const FALLBACK: [(&str, &str); 8] = [
    ("base10", "base00"),
    ("base11", "base00"),
    ("base12", "base08"),
    ("base13", "base0A"),
    ("base14", "base0B"),
    ("base15", "base0C"),
    ("base16", "base0D"),
    ("base17", "base0E"),
];

/// Reads theme `id` from a base24 or base16 YAML `text`: the D2 mapping, then the file's `kb:` section.
///
/// # Errors
///
/// The text is not YAML, lacks one of base00–base0F, or holds a value that is not a colour.
pub fn parse(id: &str, text: &str) -> Result<Scheme, SchemeError> {
    let node = kb_domain::yaml::load(text)
        .map_err(|e| SchemeError::Yaml(e.to_string()))?
        .ok_or_else(|| SchemeError::Yaml("empty".into()))?;
    let json = kb_domain::yaml::to_json(&node);
    let colour = |key: String, value: &serde_json::Value| -> Result<Rgba8, SchemeError> {
        let text = value.as_str().map_or_else(|| value.to_string(), str::to_owned);
        Rgba8::parse(&text).ok_or(SchemeError::BadColour { key, value: text })
    };
    let palette = &json["palette"];
    let base = |name: &str| -> Result<Rgba8, SchemeError> {
        match palette.get(name) {
            Some(value) => colour(format!("palette.{name}"), value),
            None => match FALLBACK.iter().find(|(k, _)| *k == name) {
                Some((_, to)) => match palette.get(*to) {
                    Some(value) => colour(format!("palette.{to}"), value),
                    None => Err(SchemeError::Missing((*to).into())),
                },
                None => Err(SchemeError::Missing(name.into())),
            },
        }
    };
    for i in 0..16 {
        base(&format!("base{i:02X}"))?;
    }
    let b = |name: &str| base(name);
    let mut p = Palette([Rgba8 { r: 0, g: 0, b: 0, a: 255 }; 19]);
    let mapping: [(&str, Rgba8); 19] = [
        ("window", b("base00")?),
        ("badge", b("base01")?),
        ("selected", b("base02")?),
        ("hover", b("base02")?.mix(b("base00")?, 0.5)),
        ("mantle", b("base10")?),
        ("code", b("base10")?),
        ("crust", b("base11")?),
        ("text", b("base05")?),
        ("muted", b("base04")?),
        ("faint", b("base03")?),
        ("edge", b("base03")?.with_alpha(0.5)),
        ("ring", b("base05")?.with_alpha(0.08)),
        ("red", b("base08")?),
        ("peach", b("base09")?),
        ("yellow", b("base0A")?),
        ("green", b("base0B")?),
        ("pink", b("base0C")?),
        ("blue", b("base0D")?),
        ("mauve", b("base0E")?),
    ];
    for (role, value) in mapping {
        p.set(role, value);
    }
    if let Some(kb) = json.get("kb").and_then(serde_json::Value::as_object) {
        for (role, value) in kb {
            let value = colour(format!("kb.{role}"), value)?;
            if !p.set(role, value) {
                return Err(SchemeError::BadColour { key: format!("kb.{role}"), value: "not a colour role".into() });
            }
        }
    }
    let label = json.get("name").and_then(serde_json::Value::as_str).unwrap_or(id).to_owned();
    Ok(Scheme { id: id.into(), label, palette: p })
}
```

#[expect(clippy::expect_used, reason = "the shipped theme is checked by a unit test")]
pub fn native() -> Palette {

/// Writes each shipped theme into `dir` unless a theme of that id is there (`.yaml` or `.yml`).
///
/// # Errors
///
/// `dir` cannot be created or a file cannot be written.
pub fn seed(dir: &Path) -> io::Result<()> {
    fs::create_dir_all(dir)?;
    for (id, text) in SHIPPED {
        if !dir.join(format!("{id}.yaml")).exists() && !dir.join(format!("{id}.yml")).exists() {
            fs::write(dir.join(format!("{id}.yaml")), text)?;
        }
    }
    Ok(())
}

/// One theme file: its id, and the theme or why it is not one.
#[derive(Debug, Clone)]
pub struct Entry {
    /// The file stem.
    pub id: String,
    /// The theme, or the reason it is broken.
    pub scheme: Result<Scheme, String>,
}

impl Entry {
    /// The scheme's label, else the id.
    #[must_use]
    pub fn label(&self) -> &str {
        self.scheme.as_ref().map_or(self.id.as_str(), |s| s.label.as_str())
    }
}

/// Every `*.yaml` and `*.yml` in `dir`, by label; the shipped themes when `dir` cannot be read.
#[must_use]
pub fn list(dir: &Path) -> Vec<Entry> {
    let mut entries: Vec<Entry> = match fs::read_dir(dir) {
        Ok(read) => read
            .filter_map(Result::ok)
            .map(|e| e.path())
            .filter(|p| p.extension().is_some_and(|x| x == "yaml" || x == "yml"))
            .filter_map(|path| {
                let id = path.file_stem()?.to_string_lossy().into_owned();
                let scheme = fs::read_to_string(&path)
                    .map_err(|e| e.to_string())
                    .and_then(|text| parse(&id, &text).map_err(|e| e.to_string()));
                Some(Entry { id, scheme })
            })
            .collect(),
        Err(_) => SHIPPED
            .iter()
            .map(|(id, text)| Entry { id: (*id).into(), scheme: parse(id, text).map_err(|e| e.to_string()) })
            .collect(),
    };
    entries.sort_by(|a, b| a.label().to_lowercase().cmp(&b.label().to_lowercase()));
    entries
}

/// Theme `id` from `dir`, else the shipped theme of that id.
///
/// # Errors
///
/// No such theme, or its file is not a theme.
pub fn load(dir: &Path, id: &str) -> Result<Scheme, String> {
    for ext in ["yaml", "yml"] {
        if let Ok(text) = fs::read_to_string(dir.join(format!("{id}.{ext}"))) {
            return parse(id, &text).map_err(|e| format!("{id}.{ext}: {e}"));
        }
    }
    SHIPPED
        .iter()
        .find(|(shipped, _)| *shipped == id)
        .map_or_else(|| Err(format!("no theme “{id}” in {}", dir.display())), |(_, text)| parse(id, text).map_err(|e| e.to_string()))
}

pub fn resolve(dir: &Path, id: Option<&str>, default: &str, overrides: &toml::Table) -> (String, Palette, Vec<String>) {

    ///
    /// # Errors
    ///
    /// The file cannot be read, is not TOML, or cannot be written.
    pub fn save_theme(&self, id: &str) -> io::Result<()> {
        let text = match fs::read_to_string(&self.preferences) {
            Ok(text) => text,
            Err(error) if error.kind() == io::ErrorKind::NotFound => String::new(),
            Err(error) => return Err(error),
        };

        doc["theme"] = toml_edit::value(id);

            fs::create_dir_all(parent)?;
        }
        fs::write(&self.preferences, doc.to_string())
    }
